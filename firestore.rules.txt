rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isAdmin() {
      return request.auth != null && request.auth.token.email == 'admin@portal.example.com';
    }
    match /settings/{doc} {
      allow read: if true;
      allow write: if isAdmin();
    }
    match /students/{id} {
      allow read: if isAdmin() || (request.auth != null && request.auth.token.email == id + '@portal.example.com');
      allow write: if isAdmin();
    }
  }
}
